#include <Wire.h>               // Librería para comunicación I2C 
#include <LiquidCrystal_I2C.h>  // Librería para el LCD con módulo I2C
#include <SPI.h>                // Librería para comunicación SPI
#include <MFRC522.h>            // Librería para el módulo RFID
#include <Keypad.h>             // Librería para el teclado matricial
#include <SoftwareSerial.h>     // Librería para comunicación serial en otros pines
#include <Servo.h>              // Librería para el control del servo

//  Configuraciones
#define SS_PIN 10               
#define RST_PIN 9               
#define SERVO_PIN A1            

const int SECUENCIA_CORRECTA[4] = {8, 8, 2, 8}; 
byte UID_VALIDO[4] = {0x14, 0x8A, 0x32, 0x2B};  
const unsigned long DELAY_ENTRE_TARJETAS = 2000;

// Objetos
LiquidCrystal_I2C lcd(0x27, 16, 2);
MFRC522 rfid(SS_PIN, RST_PIN);
SoftwareSerial btSerial(A2, A3);
Servo servoMotor;

// Configuración del teclado
const byte FILAS = 4;  
const byte COLUMNAS = 4;  

char teclas[FILAS][COLUMNAS] = {
  {'1', '2', '3', '*'},
  {'4', '5', '6', '0'},
  {'7', '8', '9', '#'},
  {'*', '0', '#', ' '}
};

byte pinFila[FILAS] = {3, 4, 5, 6};      
byte pinColumna[COLUMNAS] = {2, 7, 8, 1}; 

Keypad teclado = Keypad(makeKeymap(teclas), pinFila, pinColumna, FILAS, COLUMNAS);

// --- Variables ---
int secuenciaIngresada[4] = {0, 0, 0, 0};
int indiceSecuencia = 0;
bool bienvenidoActivo = false;
bool servoActivo = false;
bool tarjetaPreviamenteLeida = false;
byte ultimoUIDLeido[4] = {0, 0, 0, 0};
unsigned long tiempoUltimaLectura = 0; 

void setup() {
  Serial.begin(9600);        
  btSerial.begin(9600);      
  lcd.init();                
  lcd.backlight();          
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Ingrese clave");

  SPI.begin();               
  rfid.PCD_Init();           

  servoMotor.attach(SERVO_PIN);  
  servoMotor.write(120);          
}

void loop() {
  leerTarjetaRFID();
  verificarTeclado();
  verificarBluetooth();
}

// --- Funciones ---
void leerTarjetaRFID() {
  if (!rfid.PICC_IsNewCardPresent() || !rfid.PICC_ReadCardSerial()) return;

  unsigned long tiempoActual = millis();

  if (validarUID(rfid.uid.uidByte)) {
    if (tarjetaPreviamenteLeida && compararUIDs(ultimoUIDLeido, rfid.uid.uidByte)) {
      if (tiempoActual - tiempoUltimaLectura >= DELAY_ENTRE_TARJETAS) {
        reiniciarSistema();  // Resetea el sistema
        delay(DELAY_ENTRE_TARJETAS); 
      }
    } else {
      mostrarBienvenido();
      activarServo();
      guardarUltimoUID(rfid.uid.uidByte);
      tarjetaPreviamenteLeida = true;
      tiempoUltimaLectura = tiempoActual;
      delay(DELAY_ENTRE_TARJETAS);
    }
  } else {
    mostrarError();
  }
}

bool validarUID(byte *uid) {
  for (int i = 0; i < 4; i++) {
    if (uid[i] != UID_VALIDO[i]) return false;
  }
  return true;
}

bool compararUIDs(byte *uid1, byte *uid2) {
  for (int i = 0; i < 4; i++) {
    if (uid1[i] != uid2[i]) return false;
  }
  return true;
}

void guardarUltimoUID(byte *uid) {
  for (int i = 0; i < 4; i++) {
    ultimoUIDLeido[i] = uid[i];
  }
}

void verificarTeclado() {
  char tecla = teclado.getKey();
  if (!tecla) return;

  if (tecla == '0' && bienvenidoActivo) {
    reiniciarSistema();
    return;
  }

  if (tecla == '*' && indiceSecuencia == 4) {
    if (validarSecuencia()) {
      mostrarBienvenido();
      activarServo();
    } else {
      mostrarError();
    }
    reiniciarSecuencia();
  } else if (tecla >= '0' && tecla <= '9') {
    if (indiceSecuencia < 4) {
      secuenciaIngresada[indiceSecuencia++] = tecla - '0';
      mostrarClaveIngresada();
    }
  }
}

bool validarSecuencia() {
  for (int i = 0; i < 4; i++) {
    if (secuenciaIngresada[i] != SECUENCIA_CORRECTA[i]) return false;
  }
  return true;
}

void mostrarClaveIngresada() {
  lcd.clear();
  lcd.setCursor(0, 0);
  for (int i = 0; i < indiceSecuencia; i++) {
    lcd.print(secuenciaIngresada[i]);
  }
  if (indiceSecuencia == 4) {
    lcd.setCursor(0, 1);
    lcd.print("Presione *");
  }
}

void verificarBluetooth() {
  if (btSerial.available() >= 2) {
    byte byte1 = btSerial.read();
    byte byte2 = btSerial.read();

    if (byte1 == 0x33 && byte2 == 0x99) {
      reiniciarSistema();  
      return;  
    } else if (byte1 == 0x88 && byte2 == 0x28) {
      mostrarBienvenido();
      activarServo();
    }
  }
}

void mostrarBienvenido() {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Bienvenido");
  bienvenidoActivo = true;
}

void mostrarError() {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Clave incorrecta");
  delay(2000);
  lcd.clear();
  lcd.print("Ingrese clave");
}

void activarServo() {
  servoMotor.write(60);  
  servoActivo = true;
}

void reiniciarSecuencia() {
  indiceSecuencia = 0;
  for (int i = 0; i < 4; i++) secuenciaIngresada[i] = 0;
}

void reiniciarSistema() {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Ingrese clave");
  bienvenidoActivo = false;
  if (servoActivo) {
    servoMotor.write(120);  
    servoActivo = false;
  }
  tarjetaPreviamenteLeida = false;
  reiniciarSecuencia();
}
